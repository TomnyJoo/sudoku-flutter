# 1. 引言

## 1.1 编写目的

本文档旨在对异形数独游戏集合系统进行基于 Flutter 框架的详细设计说明，重点阐明 MVVM 分层职责、接口抽象设计、可扩展性实现及关键模块规范。文档适用于开发人员、测试人员及维护人员。

## 1.2 设计目标

架构清晰：严格遵循 MVVM 分层，明确 Model、ViewModel、View 边界。

高度可扩展：通过接口与抽象基类隔离变化，支持多种异形数独。

配置化与国际化：配置文件无硬编码中文，通过 ARB 本地化。

用户体验：响应式布局、智能高亮、自动候选数模式、流畅交互。

# 2. 总体架构与分层职责

## 2.1 分层定义

层级

职责

依赖方向

Model

领域实体、核心游戏规则算法（生成、求解、验证、评估）、游戏逻辑抽象基类

无外部依赖（纯 Dart）

Service

跨模块公共服务：配置读取、设置持久化、统计记录、数据库、游戏工厂注册

可依赖 Model 层接口

ViewModel

管理 UI 状态、响应用户命令、调用 Model 与 Service，通知 View 更新

依赖 Service 与 Model

View

纯 UI 渲染（Widget），绑定 ViewModel，响应用户手势

仅依赖 ViewModel 抽象

关于算法归属：生成器核心算法、求解器、验证逻辑属于 Model 层，因为它们是业务领域核心。生成器本身可以通过 Service 层的 IGeneratorService 调用，但算法实现在 Model 层。

# 3. Model 层详细设计（核心业务逻辑）

## 3.1 抽象基类 BaseSudokuGame

定义所有数独变体必须实现的契约。命名体现功能。

dart

/// 数独游戏逻辑抽象基类

/// 所有变体（标准、杀手、锯齿等）均需实现此接口

abstract class BaseSudokuGame {

/// 游戏唯一标识（对应配置中的 id）

String get gameId;

/// 棋盘规格

int get boardRows;

int get boardCols;

/// 区域类型描述（用于 UI 高亮、绘制宫线等）

RegionType get regionType;

/// 生成符合难度参数的新题目

/// 返回的 Board 中已包含预填数字（fixed）

Future&lt;Board&gt; generatePuzzle(DifficultyParams params);

/// 验证在指定位置填入某值是否违反基本规则

bool isValidMove(Board board, int row, int col, int value);

/// 求解当前盘面，返回完整答案（若无解返回 null）

Future&lt;Board?&gt; solve(Board board);

/// 获取提示：若有指定行列则提示该格，否则提示第一个空白格

HintResult getHint(Board board, {int? row, int? col});

/// 自动计算所有空白格的候选数

/// [advanced] 为 true 时采用高级排除策略（如区块、数对）

void autoCandidates(Board board, {bool advanced = false});

/// 判断当前盘面是否已完成并正确

bool isPuzzleSolved(Board board);

/// 用于难度评估的启发式方法（可选实现）

double evaluateDifficulty(Board board) =&gt; 0.0;

/// 获取游戏专属的特殊控制按钮列表（用于 UI 动态生成）

List&lt;GameSpecificAction&gt; getSpecificActions() =&gt; [];

}

/// 游戏专属动作描述

class GameSpecificAction {

final String iconAsset;      // 图标资源

final String labelKey;       // 本地化标签键

final Function(Board) action;

final bool isToggle;         // 是否为切换状态

}

/// 通用谜题生成器（使用挖洞法）

class PuzzleGenerator {

PuzzleGenerator(this.game);

Future&lt;Board&gt; generate(DifficultyParams params) async {

// 1. 生成完整终盘（调用求解器或快速填充）

final fullBoard = await _generateFullBoard();

// 2. 根据难度参数挖洞

final puzzle = _digHoles(fullBoard, params);

// 3. 验证唯一解（调用求解器）

while (!await _hasUniqueSolution(puzzle)) {

_refillHole(puzzle);

}

return puzzle;

}

## 3.5 自动候选数状态处理

自动候选数是一种 状态标记模式，在 ViewModel 中维护一个布尔值 isAutoCandidateMode，当该模式开启时，每次执行修改盘面的操作（填数、清除、撤销/重做）后，自动调用 game.autoCandidates(board, advanced: ...)。

# 5. ViewModel 层设计（统一规范）

## 5.1 主游戏 ViewModel：GamePlayViewModel

命名统一：GamePlay 明确表示游戏对局，避免歧义。

dart

class GamePlayViewModel extends ChangeNotifier {

final IGameFactory _gameFactory;

final IRendererFactory _rendererFactory;

final ISettingsService _settingsService;

final IStatisticsService _statisticsService;

late BaseSudokuGame _currentGame;

IBoardRenderer get boardRenderer =&gt; _rendererFactory.getRenderer(_currentGame.gameId);

Board _board;

HistoryManager _history;

bool _isNoteMode = false;

bool _isAutoCandidateMode = false;   // 自动候选数模式开关

int _errorsCount = 0;

Duration _elapsedTime = Duration.zero;

Duration? _bestTime;

bool _isCompleted = false;

bool _isBusy = false;                // 生成/求解中

Timer? _timer;

// 暴露给 View 的状态 getter

Board get board =&gt; _board;

bool get isNoteMode =&gt; _isNoteMode;

bool get isAutoCandidateMode =&gt; _isAutoCandidateMode;

int get errorsCount =&gt; _errorsCount;

Duration get elapsedTime =&gt; _elapsedTime;

Duration? get bestTime =&gt; _bestTime;

bool get isCompleted =&gt; _isCompleted;

bool get isBusy =&gt; _isBusy;

bool get canUndo =&gt; _history.canUndo;

bool get canRedo =&gt; _history.canRedo;

List&lt;GameSpecificAction&gt; get specificActions =&gt; _currentGame.getSpecificActions();

// 命令处理方法

void selectCell(int row, int col) {

_board.selectedRow = row;

_board.selectedCol = col;

notifyListeners();

}

void inputNumber(int number) {

if (_isCompleted || _isBusy) return;

final row = _board.selectedRow;

final col = _board.selectedCol;

if (row == -1 || col == -1) return;

if (_board.cells[row][col].isFixed) return;

if (_isNoteMode) {

_toggleCandidate(row, col, number);

} else {

_setValue(row, col, number);

}

if (_isAutoCandidateMode) {

_currentGame.autoCandidates(_board, advanced: _settingsService.isAdvancedCandidateEnabled());

}

_checkCompletion();

notifyListeners();

}

void deleteCell() {

// ... 类似处理，最后检查自动候选数

}

void toggleAutoCandidateMode() {

_isAutoCandidateMode = !_isAutoCandidateMode;

if (_isAutoCandidateMode) {

_currentGame.autoCandidates(_board, advanced: _settingsService.isAdvancedCandidateEnabled());

}

notifyListeners();

}

void undo() {

_history.undo(_board);

if (_isAutoCandidateMode) {

_currentGame.autoCandidates(_board, advanced: _settingsService.isAdvancedCandidateEnabled());

}

notifyListeners();

}

void redo() { /* 同理 */ }

Future&lt;void&gt; hint() async {

final hint = _currentGame.getHint(_board, row: _board.selectedRow, col: _board.selectedCol);

if (hint.value != null) {

final command = SetValueCommand(hint.row, hint.col, _board.cells[hint.row][hint.col].value, hint.value);

_history.execute(command, _board);

_board.selectedRow = hint.row;

_board.selectedCol = hint.col;

}

if (_isAutoCandidateMode) {

_currentGame.autoCandidates(_board, advanced: _settingsService.isAdvancedCandidateEnabled());

}

_checkCompletion();

notifyListeners();

}

// 其他方法：solve, reset, newGame, clearCell...

void _checkCompletion() {

if (_currentGame.isPuzzleSolved(_board)) {

_timer?.cancel();

_isCompleted = true;

_statisticsService.recordGame(/* ... */);

_checkNewRecord();

notifyListeners();

}

}

@override

void dispose() {

_timer?.cancel();

super.dispose();

}

}

# 8. 扩展性示例：新增游戏类型

实现游戏逻辑：创建 KillerSudoku 继承 BaseSudokuGame，实现所有抽象方法。

实现渲染器（可选）：创建 KillerBoardRenderer 实现 IBoardRenderer，负责绘制 cage。

注册：在应用启动时调用 gameFactory.register('killer', () =&gt; KillerSudoku()) 和 rendererFactory.register('killer', () =&gt; KillerBoardRenderer())。

配置：在 game_types.json 中添加条目。完成以上步骤后，UI 自动支持新游戏。

# 9. 总结

本次设计明确了以下要点：

算法归属：核心游戏算法属于 Model 层，生成器可置于 Model/algorithms 中。

接口抽象：通过 BaseSudokuGame 统一游戏逻辑契约，通过 IBoardRenderer 抽象绘制差异。

命名规范：ViewModel 以功能命名（如 GamePlayViewModel），避免歧义。

自动候选数：作为状态标记，由 ViewModel 控制，在每次修改盘面后触发。

键盘布局：强制正方形，数字与功能键盘尺寸一致。

配置无中文：全部使用英文键，通过 ARB 本地化。

该设计保证了系统的健壮性、可测试性与可扩展性。
